#!/usr/local/bin/python3
"""Construction probe only. No robot adapter, policy or conformance result."""
import errno
import json
import os
from pathlib import Path
import resource
import socket
import subprocess
import sys

report = {'scope': 'CONSTRUCTION_ISOLATION_PROBE_ONLY', 'uid': os.getuid(), 'gid': os.getgid()}
status = dict(line.split(':',1) for line in Path('/proc/self/status').read_text().splitlines() if ':' in line)
report['kernel'] = {key: status[key].strip() for key in ('CapEff', 'NoNewPrivs', 'Seccomp')}
report['cgroup'] = {name: Path('/sys/fs/cgroup', name).read_text().strip() for name in ('cpu.max','memory.max','memory.swap.max','pids.max')}
report['rlimit_cpu'] = resource.getrlimit(resource.RLIMIT_CPU)
report['rlimit_core'] = resource.getrlimit(resource.RLIMIT_CORE)
report['rlimit_nofile'] = resource.getrlimit(resource.RLIMIT_NOFILE)
report['write_refusals'] = {}
for path in ('/opt/accretion/forbidden-probe', '/run/accretion/forbidden-probe'):
    try:
        Path(path).write_text('construction probe')
        report['write_refusals'][path] = None
    except OSError as exc:
        report['write_refusals'][path] = exc.errno
try:
    with socket.socket() as connection:
        connection.settimeout(0.2)
        connection.connect(('1.1.1.1', 443))
    report['egress_error'] = None
except OSError as exc:
    report['egress_error'] = exc.errno
report['docker_socket_present'] = Path('/var/run/docker.sock').exists()
report['physical_device_paths'] = [str(p) for pattern in ('ttyUSB*','ttyACM*','gpiochip*','spidev*') for p in Path('/dev').glob(pattern)]
try:
    Path('/tmp/quota-probe').write_bytes(b'x' * (2 * 1024 * 1024))
    report['tmp_quota_error'] = None
except OSError as exc:
    report['tmp_quota_error'] = exc.errno
finally:
    Path('/tmp/quota-probe').unlink(missing_ok=True)
child = subprocess.Popen([sys.executable, '-c', 'import time; time.sleep(60)'])
report['child_pid'] = child.pid
print(json.dumps(report), flush=True)
sys.stdin.buffer.read(1)
